CREATE TABLE IF NOT EXISTS "profiles" (
  id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  role text NOT NULL CHECK (role IN ('RH', 'EMPLOYE')),
  nom text NOT NULL,
  prenom text NOT NULL,
  service text,
  date_embauche date,
  created_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS "planning" (
  id bigserial PRIMARY KEY,
  utilisateur_id uuid REFERENCES profiles(id) ON DELETE CASCADE,
  date_debut timestamptz NOT NULL,
  date_fin timestamptz NOT NULL,
  commentaire text,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now(),
  CHECK (date_fin > date_debut)
);

CREATE INDEX IF NOT EXISTS idx_planning_user ON planning(utilisateur_id);

CREATE TABLE IF NOT EXISTS "demande_changement" (
  id bigserial PRIMARY KEY,
  utilisateur_id uuid REFERENCES profiles(id) ON DELETE CASCADE,
  type_demande text CHECK (type_demande IN ('CONGE', 'HORAIRE', 'ECHANGE')),
  motif text,
  statut text DEFAULT 'EN_ATTENTE' CHECK (statut IN ('EN_ATTENTE', 'ACCEPTEE', 'REFUSEE')),
  valide_par uuid REFERENCES profiles(id),
  date_traitement timestamptz,
  created_at timestamptz DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_demande_user ON demande_changement(utilisateur_id);

CREATE TABLE IF NOT EXISTS "alerte_burnout" (
  id bigserial PRIMARY KEY,
  utilisateur_id uuid REFERENCES profiles(id) ON DELETE CASCADE,
  niveau_urgence integer CHECK (niveau_urgence BETWEEN 1 AND 3),
  statut text DEFAULT 'NOUVEAU' CHECK (statut IN ('NOUVEAU', 'EN_COURS', 'TRAITE')),
  commentaire_rh text,
  created_at timestamptz DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_alerte_user ON alerte_burnout(utilisateur_id);

CREATE TABLE IF NOT EXISTS "questionnaire" (
  id bigserial PRIMARY KEY,
  titre text NOT NULL,
  description text,
  structure_json jsonb NOT NULL,
  created_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS "reponse_questionnaire" (
  id bigserial PRIMARY KEY,
  utilisateur_id uuid REFERENCES profiles(id) ON DELETE CASCADE,
  questionnaire_id bigint REFERENCES questionnaire(id) ON DELETE CASCADE,
  reponse_json jsonb NOT NULL,
  created_at timestamptz DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_reponse_user ON reponse_questionnaire(utilisateur_id);

CREATE TABLE IF NOT EXISTS "session_ia" (
  id bigserial PRIMARY KEY,
  utilisateur_id uuid REFERENCES profiles(id) ON DELETE CASCADE,
  resume_anonymise text,
  score_risque integer,
  created_at timestamptz DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_session_user ON session_ia(utilisateur_id);

alter table profiles enable row level security;
alter table planning enable row level security;
alter table demande_changement enable row level security;
alter table alerte_burnout enable row level security;
alter table reponse_questionnaire enable row level security;
alter table session_ia enable row level security;

create policy "User can create own profile"
on profiles
for insert
with check (auth.uid() = id);

create policy "Select own profile"
on profiles
for select
using (auth.uid() = id);

create policy "Update own profile"
on profiles
for update
using (auth.uid() = id)
with check (auth.uid() = id);

create policy "RH can read all profiles"
on profiles
for select
using (
  exists (
    select 1
    from profiles p
    where p.id = auth.uid() and p.role = 'RH'
  )
);
